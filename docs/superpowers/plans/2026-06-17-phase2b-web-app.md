`web/src/lib/api/client.ts`:
```ts
const BASE = (import.meta.env.VITE_API as string) ?? "/api/v1";
let token = localStorage.getItem("tm_token") ?? "";

export function setToken(t: string) {
  token = t;
  if (t) localStorage.setItem("tm_token", t);
  else localStorage.removeItem("tm_token");
}
export function getToken() { return token; }

export class ApiError extends Error {
  code: string;
  status: number;
  constructor(status: number, code: string, message: string) {
    super(message);
    this.code = code;
    this.status = status;
  }
}

export async function apiFetch<T = unknown>(path: string, opts: RequestInit = {}): Promise<T> {
  const res = await fetch(BASE + path, {
    ...opts,
    headers: {
      ...(opts.body && !(opts.body instanceof FormData) ? { "Content-Type": "application/json" } : {}),
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(opts.headers ?? {}),
    },
  });
  if (res.status === 204) return undefined as T;
  const body = await res.json().catch(() => ({}));
  if (!res.ok) {
    const e = body?.error ?? {};
    throw new ApiError(res.status, e.code ?? "error", e.message ?? res.statusText);
  }
  return body as T;
}

export function qs(params: Record<string, string | undefined>): string {
  const u = new URLSearchParams();
  for (const [k, v] of Object.entries(params)) if (v) u.set(k, v);
  const s = u.toString();
  return s ? `?${s}` : "";
}
```

```ts
export function fmtMoney(v: number, currency: string, locale: string): string {
  return new Intl.NumberFormat(locale, { style: "currency", currency }).format(v);
}
export function fmtSignedMoney(v: number, currency: string, locale: string): string {
  const s = fmtMoney(Math.abs(v), currency, locale);
  return v < 0 ? `-${s}` : `+${s}`;
}
export function fmtPct(ratio: number, locale: string): string {
  return new Intl.NumberFormat(locale, { style: "percent", maximumFractionDigits: 0 }).format(ratio);
}
```

- [ ] **Step 4: Implement `calendar.ts`**

```ts
export interface DayCell { date: string; pnl: number | null }
export interface MonthGrid { weeks: (DayCell | null)[][]; monthTotal: number }

// month is 1-based. Builds a fixed 6-row grid (Sun-first) with daily P&L.
export function monthGrid(year: number, month: number, pnl: Record<string, number>): MonthGrid {
  const first = new Date(Date.UTC(year, month - 1, 1));
  const startDow = first.getUTCDay();
  const daysInMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();
  const cells: (DayCell | null)[] = [];
  for (let i = 0; i < startDow; i++) cells.push(null);
  let monthTotal = 0;
  for (let d = 1; d <= daysInMonth; d++) {
    const date = `${year}-${String(month).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
    const v = pnl[date] ?? null;
    if (v != null) monthTotal += v;
    cells.push({ date, pnl: v });
  }
  while (cells.length < 42) cells.push(null);
  const weeks: (DayCell | null)[][] = [];
  for (let i = 0; i < 42; i += 7) weeks.push(cells.slice(i, i + 7));
  return { weeks, monthTotal: Math.round(monthTotal * 100) / 100 };
}
```

```ts
import { create } from "zustand";

interface FilterState {
  accountId?: string;
  from?: string;
  to?: string;
  symbol?: string;
  setAccount: (id?: string) => void;
  setRange: (from?: string, to?: string) => void;
  setSymbol: (s?: string) => void;
  reset: () => void;
  toParams: () => { account_id?: string; from?: string; to?: string; symbol?: string };
}

export const useFilters = create<FilterState>((set, get) => ({
  setAccount: (accountId) => set({ accountId }),
  setRange: (from, to) => set({ from, to }),
  setSymbol: (symbol) => set({ symbol }),
  reset: () => set({ accountId: undefined, from: undefined, to: undefined, symbol: undefined }),
  toParams: () => {
    const s = get();
    return { account_id: s.accountId, from: s.from, to: s.to, symbol: s.symbol };
  },
}));
```

Each screen task: (1) write a smoke/behavior test (Testing Library) that asserts the screen renders key data from a mocked API, (2) have `design-taste-frontend` build the screen UI to the spec + its layout family, (3) wire it to the query hooks, (4) register the route, (5) verify + commit. ### Task 8: Dashboard
**Route:** `/dashboard`. **Layout family:** overview grid. **Data:** `useSummary`, `useEquityCurve`, `useDailyPnl`, `useTrades` (recent, limited). **Contents:** KPI `StatCard`s (net P&L via `fmtSignedMoney`, win rate via `fmtPct`, profit factor, expectancy, avg R), equity-curve area chart (Recharts) from `points[]`, a mini current-month calendar (reusing `monthGrid` + daily pnl), a recent-trades table (last ~10). **Test:** mock the four endpoints; assert a KPI value and a recent-trade symbol render. ### Task 9: Calendar
**Route:** `/calendar`. **Layout family:** calendar grid. **Data:** `useDailyPnl(filters)`, `useTrades({...filters, from, to})` on day-click. **Contents:** month nav (prev/next), 6x7 heatmap via `monthGrid`; each cell shows day number + `fmtSignedMoney(pnl)`, background intensity by magnitude (use `pnlColor` + opacity ramp); weekly total column; clicking a day opens that day's trades (drawer or inline list). **Test:** mock daily pnl `{ "2026-06-15": -50 }`; assert the cell for the 15th shows the loss value. ### Task 10: Trades log

### Task 11: Trade detail
**Route:** `/trades/$id`. **Layout family:** detail/record. **Data:** `useTradeDetail(id)`, `usePatchTrade()`, `useAttachments(id)` (+upload/delete), `useSetups()`, `useTags()`. **Contents:** header (symbol/dir/net P&L/R/dates); **fills** table from `fills[]`; **journal panel**: notes textarea (debounced PATCH), setup `<select>` from setups, tag multiselect (custom tags) + mistakes multiselect (kind=mistake), initial-risk number input (PATCH → recomputes R server-side); **screenshots**: upload (drag/file, image-only) calling the attachments API, gallery with delete, click → `/attachments/:id/file`. **Test:** mock the detail endpoint with notes+setup+r_multiple+2 fills; assert notes value, setup name, R, and 2 fills render; simulate editing notes → assert PATCH called. ### Task 12: Playbook
**Route:** `/playbook`. **Layout family:** stat cards/list. **Data:** `useSetups()` (+create/update/delete), `useBreakdown("setup", filters)`. **Contents:** list/grid of setups, each showing aggregate stats (trades, win rate, net P&L, profit factor, avg R) joined from the breakdown response by setup name; create/edit/delete setup dialog; click a setup → filtered trades. **Test:** mock setups + breakdown; assert a setup card shows its net P&L. ### Task 14: Settings + Import wizard
